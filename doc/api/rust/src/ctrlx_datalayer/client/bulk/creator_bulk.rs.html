7use chrono::{DateTime, Utc};
8use log::info;
9
10use crate::bindings::{
11    DLR_BULK, DLR_bulkCreate, DLR_bulkDelete, DLR_bulkGetCount, DLR_bulkGetResponseAddress,
12    DLR_bulkGetResponseData, DLR_bulkGetResponseResult, DLR_bulkGetResponseTimestamp,
13    DLR_bulkSetRequestAddress, DLR_bulkSetRequestData,
14};
15use crate::client::bulk::{BulkResponse, Requests};
16use crate::variant::dl_variant::{CVariant, filetime_to_datetime_utc};
17
18/// RAII wrapper around a native `DLR_BULK` handle.
19///
20/// Allocates a bulk operation via the native API on construction and
21/// frees it automatically when dropped.
22pub(crate) struct BulkCreator {
23    bulk: DLR_BULK,
24}
25
26impl BulkCreator {
27    /// Creates a new bulk handle and populates it with the given requests.
28    ///
29    /// Sets both the address and data for each request index before returning.
30    /// Returns an error if any native call fails.
31    pub(crate) fn new(requests: &[Requests]) -> Result<Self, crate::Result> {
32        let bulk_handle = unsafe { DLR_bulkCreate(requests.len()) };
33        let bulk = BulkCreator { bulk: bulk_handle };
34        for (i, req) in requests.iter().enumerate() {
35            let result = unsafe {
36                DLR_bulkSetRequestAddress(bulk_handle, i, req.address.as_ptr() as *mut ::std::os::raw::c_char)
37            };
38            if result != 0 {
39                return Err(crate::Result::from(result));
40            }
41            if req.value.is_none() {
42                continue; // Skip setting data if no argument is provided
43            }
44            let result = unsafe { DLR_bulkSetRequestData(bulk_handle, i, req.value.as_ref().unwrap().get_handle()) };
45            if result != 0 {
46                return Err(crate::Result::from(result));
47            }
48        }
49        Ok(bulk)
50    }
51
52    /// Collects all responses from the completed bulk operation into a `Vec`.
53    ///
54    /// Must only be called after the bulk operation has been executed by the client.
55    pub(crate) fn create_responses(&self) -> Vec<BulkResponse> {
56        (0..self.count())
57            .map(|i| BulkResponse {
58                address: self.response_address(i),
59                value: self.response_value(i),
60                result: self.response_result(i),
61                time: self.response_time(i),
62            })
63            .collect()
64    }
65
66    /// Returns the underlying native `DLR_BULK` handle.
67    pub(crate) fn get_handle(&self) -> DLR_BULK {
68        self.bulk
69    }
70
71    /// Returns the number of entries in this bulk operation.
72    fn count(&self) -> usize {
73        unsafe { DLR_bulkGetCount(self.bulk) }
74    }
75
76    /// Returns the response address string for the entry at `index`.
77    fn response_address(&self, index: usize) -> String {
78        unsafe {
79            let c_str = DLR_bulkGetResponseAddress(self.bulk, index);
80            std::ffi::CStr::from_ptr(c_str)
81                .to_string_lossy()
82                .into_owned()
83        }
84    }
85
86    /// Returns the response value for the entry at `index`.
87    fn response_value(&self, index: usize) -> crate::Value {
88        let handle = unsafe { DLR_bulkGetResponseData(self.bulk, index) };
89        let cv = CVariant::from_handle(handle);
90        crate::Value::from(&cv)
91    }
92
93    /// Returns the result code for the entry at `index`.
94    fn response_result(&self, index: usize) -> crate::Result {
95        let res = unsafe { DLR_bulkGetResponseResult(self.bulk, index) };
96        crate::Result::from(res)
97    }
98
99    /// Returns the server-side timestamp for the entry at `index` as UTC.
100    fn response_time(&self, index: usize) -> DateTime<Utc> {
101        let ts = unsafe { DLR_bulkGetResponseTimestamp(self.bulk, index) };
102        filetime_to_datetime_utc(ts)
103    }
104}
105
106impl Drop for BulkCreator {
107    fn drop(&mut self) {
108        info!("BulkCreator dropped");
109        unsafe { DLR_bulkDelete(self.bulk) };
110    }
111}
