17impl TryFrom<&CVariant> for SubscriptionPropertiesT {
18    type Error = crate::Result;
19
20    /// Tries to deserialize subscription properties from a Flatbuffers-encoded [`CVariant`].
21    ///
22    /// Returns [`crate::Result::InvalidValue`] if the variant is not of type `Flatbuffers`
23    /// or if the buffer cannot be parsed.
24    fn try_from(value: &CVariant) -> Result<Self, Self::Error> {
25        if value.variant_type() != crate::VariantType::Flatbuffers {
26            return Err(crate::Result::InvalidValue);
27        }
28        let bytes = value.get_flatbuffers();
29        let props =
30            root_as_subscription_properties(&bytes).map_err(|_| crate::Result::InvalidValue)?;
31        Ok(props.unpack())
32    }
33}

35impl TryFrom<&NotifyInfoT> for crate::Value {
36    type Error = crate::Result;
37
38    fn try_from(info: &NotifyInfoT) -> Result<Self, Self::Error> {
39        let mut fbb = flatbuffers::FlatBufferBuilder::new();
40        let offset = info.pack(&mut fbb);
41        fbb.finish(offset, None);
42        let data = fbb.finished_data().to_vec();
43        let mut value = crate::Value::new_default();
44        value.flatbuffers_mut(data);
45        Ok(value)
46    }
47}

49/// Trait representing an active subscription on a provider node.
50///
51/// `Send + Sync` so implementors can be wrapped in `Arc` and called from any thread.
52/// The lifetime of the underlying handle is managed by the datalayer runtime:
53/// it is valid between the `Subscribe` and the corresponding `Unsubscribe` event.
54pub trait NodeSubscription: Send + Sync {
55    /// The subscribed node address.
56    fn address(&self) -> &str;
57    /// Unique handle ID of the underlying `DLR_SUBSCRIPTION`.
58    fn unique_id(&self) -> u64;
59    /// All node addresses currently registered with this subscription.
60    fn nodes(&self) -> Vec<String>;
61    /// Timestamp of the last published value as UTC `DateTime`.
62    fn datetime_utc(&self) -> chrono::DateTime<chrono::Utc>;
63    /// Subscription properties (id, intervals, …).
64    fn properties(&self) -> Result<SubscriptionPropertiesT, crate::Result>;
65    /// Publishes updated values for the subscribed nodes.
66    fn publish(
67        &self,
68        status: crate::Result,
69        items: &[NotifyItemPublish],
70    ) -> Result<(), crate::Result>;
71}

73/// Type alias for a boxed [`NodeSubscription`].
74pub type NodeSubscriptionBox = Box<dyn NodeSubscription>;
75
76/// Concrete, crate-internal implementation backed by a raw `DLR_SUBSCRIPTION` handle.
77///
78/// The pointer is stored as `usize` so that the struct is both `Send` and `Sync`
79/// without requiring `unsafe impl`. The datalayer runtime guarantees that the
80/// handle remains valid for the entire lifetime of this object (i.e. between
81/// the `Subscribe` callback and the corresponding `Unsubscribe` callback).
82pub(crate) struct NodeSubscriptionImpl {
83    /// The `DLR_SUBSCRIPTION` pointer stored as an integer.
84    handle: usize,
85    address: String,
86}
87
88impl NodeSubscriptionImpl {
89    /// Creates a new [`NodeSubscriptionImpl`] from a raw `DLR_SUBSCRIPTION` handle and the node address.
90    pub(crate) fn new(subscription: DLR_SUBSCRIPTION, address: String) -> Self {
91        NodeSubscriptionImpl {
92            handle: subscription as usize,
93            address,
94        }
95    }
96
97    /// Reconstructs the raw C handle for use inside `unsafe` C-API calls.
98    fn as_raw(&self) -> DLR_SUBSCRIPTION {
99        self.handle as DLR_SUBSCRIPTION
100    }
101}
102
103impl NodeSubscription for NodeSubscriptionImpl {
104    fn address(&self) -> &str {
105        &self.address
106    }
107
108    fn unique_id(&self) -> u64 {
109        self.handle as u64
110    }
111
112    fn nodes(&self) -> Vec<String> {
113        // Retrieve the subscribed node addresses from the datalayer runtime via the C-API.
114        let nodes = CVariant::new();
115        unsafe { crate::bindings::DLR_SubscriptionGetNodes(self.as_raw(), nodes.get_handle()) };
116        nodes.get_vec_string()
117    }
118
119    fn datetime_utc(&self) -> chrono::DateTime<chrono::Utc> {
120        let timestamp = unsafe { crate::bindings::DLR_SubscriptionGetTimestamp(self.as_raw()) };
121        crate::variant::dl_variant::filetime_to_datetime_utc(timestamp)
122    }
123
124    fn properties(&self) -> Result<SubscriptionPropertiesT, crate::Result> {
125        let value = unsafe { CVariant::from_raw(DLR_SubscriptionGetProps(self.as_raw()), true) };
126        SubscriptionPropertiesT::try_from(&value)
127    }
128
129    fn publish(
130        &self,
131        status: crate::Result,
132        items: &[NotifyItemPublish],
133    ) -> Result<(), crate::Result> {
134        if items.is_empty() {
135            return Ok(());
136        }
137        // `vars` keeps the CVariant wrappers alive for the duration of the C call.
138        let mut vars = Vec::<CVariant>::new();
139        let result = unsafe {
140            // Convert each NotifyItemPublish into a raw DLR_NOTIFY_ITEM by borrowing
141            // the CVariant handles.  The CVariants are pushed into `vars` so they
142            // are not dropped before DLR_SubscriptionPublish returns.
143            let mut c_items: Vec<crate::bindings::DLR_NOTIFY_ITEM> = items
144                .iter()
145                .map(|item| {
146                    let mut c_item = std::mem::zeroed::<crate::bindings::DLR_NOTIFY_ITEM>();
147                    let data = CVariant::from(item.data());
148                    c_item.data = data.get_handle();
149                    vars.push(data);
150                    let info = CVariant::from(&item.info());
151                    c_item.info = info.get_handle();
152                    vars.push(info);
153                    c_item
154                })
155                .collect();
156            crate::bindings::DLR_SubscriptionPublish(
157                self.as_raw(),
158                status.into(),
159                c_items.as_mut_ptr(),
160                c_items.len(),
161            )
162        };
163        let result = crate::Result::from(result);
164        if result == crate::Result::Ok {
165            Ok(())
166        } else {
167            Err(result)
168        }
169    }
170}

172/// A single item to be published to a subscription.
173///
174/// Holds the node's data value together with its notification metadata
175/// (address, timestamp, notify type, …).
176pub struct NotifyItemPublish {
177    data: crate::Value,
178    notify_info: NotifyInfoT,
179}
180
181impl NotifyItemPublish {
182    /// Creates a new [`NotifyItemPublish`] for the given node `address`.
183    ///
184    /// The timestamp defaults to the current UTC time and the notify type to
185    /// [`NotifyType::Data`].
186    pub fn new(address: String) -> Self {
187        let mut notify_info = NotifyInfoT::default();
188        notify_info.node = Some(address);
189        notify_info.timestamp =
190            crate::variant::dl_variant::datetime_utc_to_filetime(chrono::Utc::now());
191        notify_info.notifyType = NotifyType::Data;
192        Self {
193            data: Value::new_default(),
194            notify_info,
195        }
196    }
197
198    /// Returns a shared reference to the node's data value.
199    pub fn data(&self) -> &Value {
200        &self.data
201    }
202
203    /// Returns a mutable reference to the node's data value.
204    pub fn data_mut(&mut self) -> &mut Value {
205        &mut self.data
206    }
207
208    /// Serializes the notification metadata into a Flatbuffers-encoded [`Value`].
209    pub fn info(&self) -> Value {
210        crate::Value::try_from(&self.notify_info).unwrap_or_else(|_| Value::new_default())
211    }
212
213    /// Sets the notification timestamp (converted from UTC to FILETIME internally).
214    pub fn timestamp_mut(&mut self, timestamp: chrono::DateTime<chrono::Utc>) {
215        self.notify_info.timestamp =
216            crate::variant::dl_variant::datetime_utc_to_filetime(timestamp);
217    }
218
219    /// Sets the notification type (e.g. `Data`, `Event`, …).
220    pub fn notify_type_mut(&mut self, notify_type: NotifyType) {
221        self.notify_info.notifyType = notify_type;
222    }
223
224    /// Sets an optional event-type string for event notifications.
225    pub fn event_type_mut(&mut self, event_type: String) {
226        self.notify_info.eventType = Some(event_type);
227    }
228
229    /// Sets the monotonically increasing sequence number for ordering notifications.
230    pub fn sequence_mut(&mut self, sequence: u64) {
231        self.notify_info.sequenceNumber = sequence;
232    }
233
234    /// Sets the source name that identifies the origin of the notification.
235    pub fn source_mut(&mut self, source: String) {
236        self.notify_info.sourceName = Some(source);
237    }
