7use std::env;
8
9/// A newtype wrapper representing a datalayer connection string.
10#[derive(Debug, PartialEq, Clone)]
11pub struct ConnectionString(String);
12
13impl From<ConnectionString> for String {
14    /// Converts a [`ConnectionString`] into the inner [`String`] by consuming the wrapper.
15    fn from(c: ConnectionString) -> String {
16        c.0
17    }
18}
19
20impl From<&ConnectionString> for String {
21    /// Converts a reference to a [`ConnectionString`] into a [`String`] by cloning the inner value.
22    fn from(c: &ConnectionString) -> String {
23        c.0.clone()
24    }
25}
26
27/// Builder for creating a datalayer connection string.
28///
29/// Provides a fluent interface to configure the target address, credentials,
30/// and SSL port before constructing the final [`ConnectionString`].
31/// ```ignore
32/// let connection_string = ConnectionBuilder::new()
33///     .with_address("192.168.0.100".to_string())
34///     .with_user("admin".to_string())
35///     .with_password("xxx".to_string())
36///     .with_sslport(8443)
37///     .build();
38/// ```
39pub struct ConnectionBuilder {
40    /// Target IP address of the ctrlX device.
41    address: String,
42    /// Username for authentication.
43    user: String,
44    /// Password for authentication.
45    password: String,
46    /// SSL port used for the TCP connection.
47    sslport: u16,
48}
49
50impl ConnectionBuilder {
51    /// Creates a new `ConnectionBuilder` with default values:
52    /// - address: `192.168.1.1`
53    /// - user: `boschrexroth`
54    /// - password: `boschrexroth`
55    /// - sslport: `443`
56    pub fn new() -> Self {
57        ConnectionBuilder {
58            address: "192.168.1.1".to_string(),
59            user: "boschrexroth".to_string(),
60            password: "boschrexroth".to_string(),
61            sslport: 443,
62        }
63    }
64
65    /// Configures the builder for a virtual machine (QEMU) environment.
66    /// Sets address to `10.0.2.2` and sslport to `8443`.
67    pub fn with_ip_virtual(&mut self) -> &mut Self {
68        self.address = "10.0.2.2".to_string();
69        self.sslport = 8443;
70        self
71    }
72
73    /// Sets the target IP address.
74    /// Example: `
75    /// builder.with_address("192.168.0.100".to_string());
76    /// builder.with_address("192.168.0.100:1234".to_string());
77    pub fn with_address(&mut self, address: String) -> &mut Self {
78        self.address = address;
79        self
80    }
81
82    /// Sets the username for authentication.
83    pub fn with_user(&mut self, user: String) -> &mut Self {
84        self.user = user;
85        self
86    }
87
88    /// Sets the password for authentication.
89    pub fn with_password(&mut self, password: String) -> &mut Self {
90        self.password = password;
91        self
92    }
93
94    /// Sets the SSL port for the TCP connection.
95    pub fn with_sslport(&mut self, sslport: u16) -> &mut Self {
96        self.sslport = sslport;
97        self
98    }
99
100    /// Builds the [`ConnectionString`].
101    ///
102    /// If the `SNAP` environment variable is set (i.e. running inside a snap),
103    /// returns `"ipc://"` for local inter-process communication.
104    /// Otherwise returns a TCP connection string of the form:
105    /// `tcp://<user>:<password>@<address>?sslport=<port>`
106    pub fn build(&self) -> ConnectionString {
107        if env::var_os("SNAP").is_some() {
108            return ConnectionString("ipc://".to_string());
109        }
110        // Build the TCP connection string from the configured parameters.
111        ConnectionString(format!(
112            "tcp://{}:{}@{}?sslport={}",
113            self.user, self.password, self.address, self.sslport
114        ))
115    }
116}

118#[cfg(test)]
119mod tests {
120    use super::*;
121    use serial_test::serial;
122    
123    #[test]
124    #[serial]
125    fn new_returns_default_values() {
126        let b = ConnectionBuilder::new();
127        let url = b.build();
128        assert_eq!(url, ConnectionString("tcp://boschrexroth:boschrexroth@192.168.1.1?sslport=443".to_string()));
129
130        let b2 = ConnectionBuilder::new().build();
131        assert_eq!(b2, ConnectionString("tcp://boschrexroth:boschrexroth@192.168.1.1?sslport=443".to_string()));
132    }
133
134    #[test]
135    #[serial]
136    fn with_ip_virtual_sets_address_and_port() {
137        let mut b = ConnectionBuilder::new();
138        b.with_ip_virtual();
139        let url = b.build();
140        assert_eq!(url, ConnectionString("tcp://boschrexroth:boschrexroth@10.0.2.2?sslport=8443".to_string()));
141        let b2 = ConnectionBuilder::new().with_ip_virtual().build();
142        assert_eq!(b2, ConnectionString("tcp://boschrexroth:boschrexroth@10.0.2.2?sslport=8443".to_string()));
143    }
144
145    #[test]
146    #[serial]
147    fn with_address_overrides_address() {
148        let mut b = ConnectionBuilder::new();
149        b.with_address("192.168.0.100".to_string());
150        let url = b.build();
151        assert_eq!(url, ConnectionString("tcp://boschrexroth:boschrexroth@192.168.0.100?sslport=443".to_string()));
152
153        let b2 = ConnectionBuilder::new().with_address("192.168.0.101".to_string()).build();
154        assert_eq!(b2, ConnectionString("tcp://boschrexroth:boschrexroth@192.168.0.101?sslport=443".to_string()));
155    }
156
157    #[test]
158    #[serial]
159    fn with_user_overrides_user() {
160        let mut b = ConnectionBuilder::new();
161        b.with_user("admin".to_string());
162        let url = b.build();
163        assert_eq!(url, ConnectionString("tcp://admin:boschrexroth@192.168.1.1?sslport=443".to_string()));
164    }
165
166    #[test]
167    #[serial]
168    fn with_password_overrides_password() {
169        let mut b = ConnectionBuilder::new();
170        b.with_password("secret".to_string());
171        let url = b.build();
172        assert_eq!(url, ConnectionString("tcp://boschrexroth:secret@192.168.1.1?sslport=443".to_string()));
173    }
174
175    #[test]
176    #[serial]
177    fn with_sslport_overrides_port() {
178        let mut b = ConnectionBuilder::new();
179        b.with_sslport(8443);
180        let url = b.build();
181        assert_eq!(url, ConnectionString("tcp://boschrexroth:boschrexroth@192.168.1.1?sslport=8443".to_string()));
182    }
183
184    #[test]
185    #[serial]
186    fn chained_setters_build_correct_url() {
187        let mut b = ConnectionBuilder::new();
188        b.with_address("10.10.10.1".to_string())
189            .with_user("user1".to_string())
190            .with_password("pass1".to_string())
191            .with_sslport(9443);
192        let url = b.build();
193        assert_eq!(url, ConnectionString("tcp://user1:pass1@10.10.10.1?sslport=9443".to_string()));
194    }
195
196    #[test]
197    #[serial]
198    fn xbuild_returns_ipc_when_snap_is_set() {
199        unsafe {
200            env::set_var("SNAP", "/snap/myapp/1");
201            let b = ConnectionBuilder::new();
202            let url = b.build();
203            env::remove_var("SNAP");
204            assert_eq!(url, ConnectionString("ipc://".to_string()));
205        }
206    }
